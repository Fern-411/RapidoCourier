    @Transactional
    public PaqueteResponse registrarPaquete(PaqueteRequest request) {
        UUID remitenteId = validarDniYObtenerId(request.dniRemitente());
        UUID destinatarioId = validarDniYObtenerId(request.dniDestinatario());

        Set<Categoria> categorias = new HashSet<>();
        for (String nombreCat : request.categorias()) {
            Categoria cat = categoriaRepository.findByNombre(nombreCat)
                    .orElseGet(() -> categoriaRepository.save(new Categoria(nombreCat)));
            categorias.add(cat);
        }

        BigDecimal tarifa = calcularTarifa(request.pesoKg(), request.valorDeclarado());

        Paquete paquete = new Paquete();
        paquete.setCodigoRastreo(UUID.randomUUID().toString().substring(0, 8).toUpperCase());
        paquete.setPesoKg(request.pesoKg());
        paquete.setValorDeclarado(request.valorDeclarado());
        paquete.setTarifa(tarifa);
        paquete.setSucursalOrigen(request.sucursalOrigen());
        paquete.setSucursalDestino(request.sucursalDestino());
        paquete.setRemitenteId(remitenteId);
        paquete.setDestinatarioId(destinatarioId);
        paquete.setCategorias(categorias);
        paquete.setEstadoActual(EstadoPaquete.REGISTRADO);

        Paquete guardado = paqueteRepository.save(paquete);

        HistorialEstado historial = new HistorialEstado(guardado, EstadoPaquete.REGISTRADO, "SISTEMA");
        historialEstadoRepository.save(historial);

        notificarCambioEstado(guardado.getId(), "Paquete registrado con codigo: " + guardado.getCodigoRastreo());

        return new PaqueteResponse(guardado.getId(), guardado.getCodigoRastreo(), guardado.getTarifa(), guardado.getEstadoActual(), guardado.getCreatedAt());
    }

    @CircuitBreaker(name = "clientesCB", fallbackMethod = "fallbackValidarDni")
    private UUID validarDniYObtenerId(String dni) {
        ApiResponse<ClienteResponse> response = clienteClient.buscarPorDni(dni);
        if (response != null && response.success() && response.data() != null) {
            return response.data().id();
        }
        throw new BaseException(ErrorCode.DNI_NO_REGISTRADO);
    }

    private UUID fallbackValidarDni(String dni, Throwable t) {
        throw new BaseException(ErrorCode.ERROR_COMUNICACION_CLIENTES);
    }

    private BigDecimal calcularTarifa(BigDecimal pesoKg, BigDecimal valorDeclarado) {
        BigDecimal tarifaBase = new BigDecimal("10.00");
        BigDecimal costoPeso = pesoKg.multiply(new BigDecimal("5.00"));
        BigDecimal seguro = BigDecimal.ZERO;
        if (valorDeclarado.compareTo(new BigDecimal("100.00")) > 0) {
            seguro = valorDeclarado.multiply(new BigDecimal("0.02"));
        }
        return tarifaBase.add(costoPeso).add(seguro);
    }

    @Transactional
    public PaqueteResponse actualizarEstado(UUID paqueteId, EstadoPaquete nuevoEstado, String usuario) {
        Paquete paquete = paqueteRepository.findById(paqueteId)
                .orElseThrow(() -> new BaseException(ErrorCode.PAQUETE_NO_ENCONTRADO));

        validarTransicionEstado(paquete.getEstadoActual(), nuevoEstado);

        if (nuevoEstado == EstadoPaquete.EN_TRANSITO) {
            validarPago(paqueteId);
        }

        paquete.setEstadoActual(nuevoEstado);
        paqueteRepository.save(paquete);

        HistorialEstado historial = new HistorialEstado(paquete, nuevoEstado, usuario);
        historialEstadoRepository.save(historial);

        notificarCambioEstado(paqueteId, "El paquete ahora esta en estado: " + nuevoEstado);

        return new PaqueteResponse(paquete.getId(), paquete.getCodigoRastreo(), paquete.getTarifa(), paquete.getEstadoActual(), paquete.getCreatedAt());
    }

    private void validarTransicionEstado(EstadoPaquete actual, EstadoPaquete nuevo) {
        boolean valido = false;
        if (actual == EstadoPaquete.REGISTRADO && nuevo == EstadoPaquete.EN_TRANSITO) valido = true;
        if (actual == EstadoPaquete.EN_TRANSITO && nuevo == EstadoPaquete.EN_SUCURSAL_DESTINO) valido = true;
        if (actual == EstadoPaquete.EN_SUCURSAL_DESTINO && nuevo == EstadoPaquete.ENTREGADO) valido = true;

        if (!valido) {
            throw new BaseException(ErrorCode.TRANSICION_INVALIDA);
        }
    }

    private void validarPago(UUID paqueteId) {
        try {
            ApiResponse<Boolean> response = pagosClient.verificarPagoCompletado(paqueteId);
            if (response == null || !response.success() || Boolean.FALSE.equals(response.data())) {
                throw new BaseException(ErrorCode.PAGO_NO_COMPLETADO);
            }
        } catch (BaseException e) {
            throw e;
        } catch (Exception e) {
            throw new BaseException(ErrorCode.ERROR_COMUNICACION_PAGOS);
        }
    }
